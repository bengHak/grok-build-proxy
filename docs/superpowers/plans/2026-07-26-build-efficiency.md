# Build Efficiency and Release Integrity Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Preserve Cargo build artifacts, remove duplicate CI/release builds, use one Cargo version source, and prevent release asset replacement.

**Architecture:** Make owns incremental-safe local packaging, while a reusable GitHub Actions quality workflow owns the shared validation and archive build contract. CI calls it directly; tag-only release performs a cheap preflight, calls the same workflow with artifact upload enabled, then publishes exactly that validated artifact without rebuilding or overwriting.

**Tech Stack:** GNU/BSD Make, Cargo/Rust 1.88, Bash, GitHub Actions reusable workflows, `gh`, macOS `tar` and `shasum`.

## Global Constraints

- macOS remains the only supported runtime and release platform.
- Rust MSRV remains exactly 1.88.
- `Cargo.toml` `[package].version` is the only project version source.
- Releases trigger only from `v*` tag pushes and require exact `v${package.version}` equality.
- An existing GitHub release is an error; no repair, upload, or overwrite path is allowed.
- Ordinary `build` and `dist` targets must never run `cargo clean`; explicit `make clean` remains destructive.
- Published bytes must be downloaded from the reusable quality workflow artifact and must not be rebuilt in the publish job.
- All GitHub Actions are pinned to full commit SHAs.
- Do not push, create tags, publish releases, or otherwise modify shared remote state.
- Do not modify runtime proxy behavior or dependencies.

- Create `.github/workflows/quality.yml`: reusable shared formatting, linting, testing, installer, distribution, archive validation, and optional artifact upload contract.

- [ ] **Step 1: Write the failing build contract script**

Create `scripts/check-build-contract.sh` with:

```sh
#!/bin/sh
set -eu

fail() {
  printf 'build contract failed: %s\n' "$*" >&2
  exit 1
}

package_version=$(cargo metadata --locked --no-deps --format-version 1 \
  | python3 -c 'import json,sys; data=json.load(sys.stdin); matches=[p["version"] for p in data["packages"] if p["name"] == "grok-build-proxy"]; assert len(matches) == 1; print(matches[0])')

[ ! -e VERSION ] || fail 'VERSION must not exist; Cargo.toml is authoritative'
[ ! -e build.rs ] || fail 'build.rs version override must not exist'

if git grep -n -E 'GROK_BUILD_PROXY_BUILD_VERSION|GROK_BUILD_PROXY_VERSION' -- ':!install.sh' ':!README.md' ':!docs/**' ':!scripts/check-build-contract.sh'; then
  fail 'build-time version override remains'
fi

grep -q 'const VERSION: &str = env!("CARGO_PKG_VERSION");' src/main.rs \
  || fail 'CLI must use CARGO_PKG_VERSION'

make_version=$(make --no-print-directory print-version)
[ "$make_version" = "$package_version" ] \
  || fail "Make version $make_version differs from Cargo version $package_version"

printf 'build contract ok: version %s\n' "$package_version"
```

BINARY := grok-build-proxy
CARGO := cargo
DIST_DIR := dist
PACKAGE_VERSION := $(shell $(CARGO) metadata --locked --no-deps --format-version 1 | python3 -c 'import json,sys; data=json.load(sys.stdin); print(next(p["version"] for p in data["packages"] if p["name"] == "$(BINARY)"))')

build:
	$(CARGO) build --locked --release
	mkdir -p bin
	cp target/release/$(BINARY) bin/$(BINARY)

build-arm64:
	rustup target add aarch64-apple-darwin
	$(CARGO) build --locked --release --target aarch64-apple-darwin
	mkdir -p bin
	cp target/aarch64-apple-darwin/release/$(BINARY) bin/$(BINARY)-darwin-arm64

build-amd64:
	rustup target add x86_64-apple-darwin
	$(CARGO) build --locked --release --target x86_64-apple-darwin
	mkdir -p bin
	cp target/x86_64-apple-darwin/release/$(BINARY) bin/$(BINARY)-darwin-amd64

prepare-dist:
	rm -rf bin $(DIST_DIR)
	mkdir -p bin $(DIST_DIR)

dist: prepare-dist build-arm64 build-amd64
	mkdir -p $(DIST_DIR)/arm64 $(DIST_DIR)/amd64
	cp bin/$(BINARY)-darwin-arm64 $(DIST_DIR)/arm64/$(BINARY)
	cp bin/$(BINARY)-darwin-amd64 $(DIST_DIR)/amd64/$(BINARY)
	cp LICENSE README.md $(DIST_DIR)/arm64/
	cp LICENSE README.md $(DIST_DIR)/amd64/
	tar -C $(DIST_DIR)/arm64 -czf $(DIST_DIR)/$(BINARY)_Darwin_arm64.tar.gz $(BINARY) LICENSE README.md
	tar -C $(DIST_DIR)/amd64 -czf $(DIST_DIR)/$(BINARY)_Darwin_amd64.tar.gz $(BINARY) LICENSE README.md
	cd $(DIST_DIR) && shasum -a 256 $(BINARY)_Darwin_arm64.tar.gz $(BINARY)_Darwin_amd64.tar.gz > checksums.txt
	rm -rf $(DIST_DIR)/arm64 $(DIST_DIR)/amd64

verify-dist:

	@test "$$(tar -tzf $(DIST_DIR)/$(BINARY)_Darwin_arm64.tar.gz | LC_ALL=C sort)" = "$$(printf '%s\n' $(BINARY) LICENSE README.md | LC_ALL=C sort)"

print-version:
	@printf '%s\n' $(PACKAGE_VERSION)

test:
	$(CARGO) test --locked --all-targets

lint:
	$(CARGO) clippy --locked --all-targets --all-features -- -D warnings

fmt:
	$(CARGO) fmt

check:
	$(CARGO) fmt --check
	$(CARGO) clippy --locked --all-targets --all-features -- -D warnings
	$(CARGO) test --locked --all-targets
	sh -n install.sh
	./scripts/check-build-contract.sh

clean:
	$(CARGO) clean
	rm -rf bin $(DIST_DIR)

- [ ] **Step 3: Replace release workflow with preflight, reusable quality, and publish jobs**

Replace `.github/workflows/release.yml`:

```yaml
name: Release

on:
  push:
    tags:
      - "v*"

permissions:
  contents: read

concurrency:
  group: release-${{ github.ref }}
  cancel-in-progress: false

jobs:
  preflight:
    runs-on: macos-14
    permissions:
      contents: read
    outputs:
      tag: ${{ steps.version.outputs.tag }}
    steps:
      - uses: actions/checkout@11d5960a326750d5838078e36cf38b85af677262 # v4
        with:
          fetch-depth: 0
      - uses: dtolnay/rust-toolchain@2c7215f132e9ebf062739d9130488b56d53c060c # master, 2026-07-16
        with:
          toolchain: 1.88.0
      - name: Verify tag and Cargo package version
        id: version
        shell: bash
        run: |
          set -euo pipefail
          version="$(cargo metadata --locked --no-deps --format-version 1 | python3 -c 'import json,sys; data=json.load(sys.stdin); print(next(p["version"] for p in data["packages"] if p["name"] == "grok-build-proxy"))')"
          expected="v${version}"
          if [[ "${GITHUB_REF_TYPE}" != "tag" || "${GITHUB_REF_NAME}" != "${expected}" ]]; then
            echo "Release tag ${GITHUB_REF_NAME} does not match Cargo package version ${expected}" >&2
            exit 1
          fi
          echo "tag=${expected}" >> "${GITHUB_OUTPUT}"
      - name: Reject an existing release
        env:
          GH_TOKEN: ${{ github.token }}
          TAG: ${{ steps.version.outputs.tag }}
        shell: bash
        run: |
          set -euo pipefail
          if gh release view "${TAG}" >/dev/null 2>&1; then
            echo "Release ${TAG} already exists; refusing to replace published assets." >&2
            exit 1
          fi

  quality:
    needs: preflight
    permissions:
      contents: read
    uses: ./.github/workflows/quality.yml
    with:
      upload_dist: true

  publish:
    needs: [preflight, quality]
    runs-on: macos-14
    permissions:
      contents: write
    steps:
      - uses: actions/download-artifact@d3f86a106a0bac45b974a628896c90dbdf5c8093 # v4
        with:
          name: macos-dist
          path: dist
      - name: Validate downloaded release archives
        shell: bash
        run: |
          set -euo pipefail
          expected=(
            dist/grok-build-proxy_Darwin_arm64.tar.gz
            dist/grok-build-proxy_Darwin_amd64.tar.gz
            dist/checksums.txt
          )
          for path in "${expected[@]}"; do
            [[ -f "${path}" ]] || { echo "Missing release asset: ${path}" >&2; exit 1; }
          done
          [[ "$(find dist -type f | wc -l | tr -d ' ')" == "3" ]] || {
            echo "Unexpected files in downloaded release artifact" >&2
            find dist -type f -print >&2
            exit 1
          }
          (cd dist && shasum -a 256 -c checksums.txt)
      - name: Publish immutable GitHub release
        env:
          GH_TOKEN: ${{ github.token }}
          TAG: ${{ needs.preflight.outputs.tag }}
        shell: bash
        run: |
          set -euo pipefail
          gh release create "${TAG}" \
            dist/grok-build-proxy_Darwin_arm64.tar.gz \
            dist/grok-build-proxy_Darwin_amd64.tar.gz \
            dist/checksums.txt \
            --generate-notes \
            --latest \
            --title "${TAG}" \
            --verify-tag
```
